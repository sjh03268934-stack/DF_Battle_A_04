<%@ page language="java" contentType="text/html; charset=UTF-8" pageEncoding="UTF-8"%>
<%@ page import="com.game.system.BattleSystem" %>
<%
    // 1. 세션에서 게임 시스템(BattleSystem) 객체 가져오기
    // (캐릭터 생성 UI에서 세션에 저장해둔 객체를 사용해야 생성된 캐릭터로 공격할 수 있습니다)
    BattleSystem battleSystem = (BattleSystem) session.getAttribute("battleSystem");
    
    // 테스트를 위해 세션에 시스템이 없으면 새로 생성 (실제로는 캐릭터 생성에서 만들어져야 함)
    if (battleSystem == null) {
        battleSystem = new BattleSystem();
        session.setAttribute("battleSystem", battleSystem);
    }

    String resultMessage = "";
    
    // 2. 몬스터 공격 요청 처리 (POST 요청일 때 동작)
    if ("POST".equalsIgnoreCase(request.getMethod())) {
        String inputId = request.getParameter("playerId");
        
        // 3. 전투 시스템으로 공격 요청 및 결과 반환
        if (inputId != null && !inputId.trim().isEmpty()) {
            resultMessage = battleSystem.attackMonster(inputId);
        } else {
            resultMessage = "플레이어 ID를 입력해주세요.";
        }
    }
%>

        <% if (!resultMessage.isEmpty()) { %>
            <div class="result-box">

            </div>
        <% } %>

        <!-- [공격버튼클릭()] 몬스터 공격 입력 폼 -->
